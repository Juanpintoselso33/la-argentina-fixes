namespace = salons

# on action every month
salons.1 = {
	type = country_event
	placement = ROOT

	title = salons.1.t
	desc = salons.1.d
	flavor = salons.1.f

	cooldown = { years = 999 }
	duration = 3

	event_image = {
		video = "southamerica_salon"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/tutorial_icon.dds"

	trigger = {
		c:ARG ?= this # important because c:ARG can not exist
		game_date >= 1837.2.1 #first day
		OR = {
			has_law = law_type:law_autocracy
			has_law = law_type:law_military_junta
			has_law = law_type:law_oligarchy
		}
		ig:ig_intelligentsia = {
			is_marginal = no
		}
	}

	immediate = {



	}

	option = {
		name = salons.1.a
		default_option = no
		ai_chance = {
			value = 4
		}

		add_modifier = {
			name = salons_neutral
			days = long_modifier_time
		}

	}

	option = {
		name = salons.1.b
		default_option = yes
		ai_chance = {
			value = 6
		}

		every_scope_character = {
			limit = {
				OR = {
					has_template = ARG_Esteban_Echeverria
				}
			}
			transfer_character = c:URU
			add_character_role = agitator
		}

		add_modifier = {
			name = salons_closed
			days = long_modifier_time
		}

	}
	}
