namespace = mazorca

# on action every month
mazorca.1 = {
	type = country_event
	placement = ROOT

	title = mazorca.1.t
	desc = mazorca.1.d
	flavor = mazorca.1.f

	cooldown = { years = 999 }
	duration = 3

	event_image = {
		video = "southamerica_aristocrats2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	trigger = {
		c:ARG ?= this # important because c:ARG can not exist

		OR = {
			has_law = law_type:law_autocracy
			has_law = law_type:law_military_junta
			has_law = law_type:law_oligarchy
		}
		has_law = law_type:law_local_police #Mazorca

		OR = {
			country_turmoil >= 0.13
			radical_population_fraction >= 0.15
			has_revolution = yes
		}
	}


	immediate = {
	}

	option = {
		name = mazorca.1.a
		default_option = yes

		add_modifier = {
			name = divisa_punzo_modifier
		}
	}
	}
