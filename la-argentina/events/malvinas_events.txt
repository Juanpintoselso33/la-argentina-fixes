	placement = ROOT

	title = malvinas.1.t

	duration = 3

		video = "unspecific_signed_contract"
	}

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	trigger = {
		OR = {
			c:Great_ARG ?= this
			c:ARG ?= this
			c:PLT ?= this
		}
	}

	immediate = {
	}

	option = {
		name = malvinas.1.a
		ai_chance = {
			base = 1
		}
		custom_tooltip = malvinas.1.tt1 # 3 million over a year
		c:GBR = {
			trigger_event = {
				id = malvinas.2
				days = 0
				popup = yes
			}
		}
	}
	option = { # perhaps another time
		name = malvinas.1.c
		default_option = yes
		ai_chance = {
			base = 0
		}
		remove_variable = malvinas_purchase_var
	}
}


# An Offer for malvinas
malvinas.2 = {
	type = country_event
	placement = ROOT

	title = malvinas.2.t
	desc = malvinas.2.d
	flavor = malvinas.2.f

	duration = 3

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	trigger = {
	c:GBR ?= this
	}

	immediate = {
	}

	option = {
		name = malvinas.2.a
		default_option = yes
		
		c:ARG ?= { trigger_event = { id = malvinas.3 days = 0 popup = yes }}
		
		c:Great_ARG ?= { trigger_event = { id = malvinas.3 days = 0 popup = yes }}
		
		c:PLT ?= { trigger_event = { id = malvinas.3 days = 0 popup = yes }}
		ai_chance = {
			base = 1
		}
	}
	option = {
		name = malvinas.2.b
		
		c:ARG ?= { trigger_event = { id = malvinas.4 days = 0 popup = yes }}
		
		c:Great_ARG ?= { trigger_event = { id = malvinas.4 days = 0 popup = yes }}
		
		c:PLT ?= { trigger_event = { id = malvinas.4 days = 0 popup = yes }}
		show_as_tooltip = {
			change_relations = {
				country = ROOT
				value = -100
			}
		}
		ai_chance = {
			base = 0
		}
	}
}
